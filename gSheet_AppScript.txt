/**
 * Google Apps Script (GS) for running time-based email campaigns.
 *
 * This script reads data only from the "Campaigns" sheet, compares the current time
 * to the scheduled send times for ACTIVE or UPCOMING campaigns, and triggers
 * an email function to a fixed recipient if a time match is found, respecting
 * the Campaign Message Count (1, 2, .., 10).
 *
 * Uses the Properties Service to track sent status (CampaignID + Template # + Date)
 * to prevent duplicate emails without modifying the source sheet.
 */

// --- GLOBAL CONFIGURATION ---

// --- PROPERTIES SERVICE FOR SENT TRACKING (NO SHEET MODIFICATION) ---

// Properties service instance for tracking sent status
const SCRIPT_PROPERTIES = PropertiesService.getScriptProperties();

// --- doGet SENT STATUS TRACKING FUNTIONS ---

/**
 * Generates a unique key for tracking the sent status of a specific message today.
 * Key format: CAMPAIGN_ID-MESSAGE_NUMBER-YYYY-MM-DD
 * @param {string} campaignId The ID of the campaign.
 * @param {number} messageNumber The template number (1, 2, or 3).
 * @returns {string} The unique key.
 */
function getSentKey(campaignId, messageNumber) {
    const today = new Date();
    // Use the spreadsheet's timezone to ensure the "day" is consistent with the scheduled time
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const formattedDate = Utilities.formatDate(today, ss.getSpreadsheetTimeZone(), 'yyyy-MM-dd');
    return `${campaignId}-${messageNumber}-${formattedDate}`;
}

/**
 * Checks if a specific campaign message has already been sent today.
 * @param {string} campaignId The ID of the campaign.
 * @param {number} messageNumber The template number.
 * @returns {boolean} True if the key exists in properties, meaning it was sent today.
 */
function wasSentToday(campaignId, messageNumber) {
    const key = getSentKey(campaignId, messageNumber);
    return SCRIPT_PROPERTIES.getProperty(key) === 'SENT';
}

/**
 * Sets a flag in the Properties Service indicating the campaign message was sent today.
 * The property value is simply 'SENT', and it persists indefinitely, but the key is date-specific.
 * @param {string} campaignId The ID of the campaign.
 * @param {number} messageNumber The template number.
 */
function setSentLog(campaignId, messageNumber) {
    const key = getSentKey(campaignId, messageNumber);
    SCRIPT_PROPERTIES.setProperty(key, 'SENT');
    Logger.log('Set sent log for key: %s', key);
}

/**
 * Reads data from a specified sheet in the active spreadsheet and converts it
 * into an array of JavaScript objects, where keys are derived from the header row.
 *
 * @param {string} sheetName The name of the sheet to read (e.g., 'Campaigns').
 * @returns {Array<Object>} An array of objects representing the sheet data.
 */
function getDataFromSheet(sheetName) {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName(sheetName);
    if (!sheet) {
        Logger.log('Error: Sheet "%s" not found in the spreadsheet.', sheetName);
        return [];
    }

    const range = sheet.getDataRange();
    const values = range.getValues();

    if (values.length < 2) {
        Logger.log('Sheet "%s" contains no data rows besides the header.', sheetName);
        return [];
    }

    const headers = values[0];
    const data = values.slice(1);
    const result = [];

    // Helper to sanitize header: removes spaces/special chars to create valid keys
    // const sanitizeHeader = (header) => header.trim().replace(/[^a-zA-Z0-9]/g, '');

    // Map sanitized headers to their original index
    const keyMap = headers.map(header => header); //sanitizeHeader(header));

    for (let i = 0; i < data.length; i++) {
        const row = data[i];
        const rowObject = { __rowIndex: i + 1 }; // Store original 1-based index for potential future updates

        for (let j = 0; j < row.length; j++) {
            const headerKey = keyMap[j];
            if (headerKey) {
                rowObject[headerKey] = row[j];
            }
        }
        result.push(rowObject);
    }
    return result;
}

/**
 * Fetches customer chat data from the "Chats" sheet.
 * Each row represents one Customer Chat.
 *
 * @returns {Array<Object>} Array of customer objects:
 * [
 *   { id, name, email, phone, category }
 * ]
 */
function getCustomerChatsData() {
  const rawData = getDataFromSheet('Chats');
  return rawData.map(row => ({
    id: row['Customer_ID'] || '',
    name: row['Customer_Name'] || '',
    email: row['Customer_Email'] || '',
    phone: row['Customer_Phone'] || '',
    category: row['Customer_Chat_Category'] || ''
  })).filter(c => c.category); // only keep valid categories
}

/**
 * Fetches campaign data from the "Campaigns" sheet.
 * Each campaign may contain up to five templates and timing pairs.
 *
 * @returns {Array<Object>} Array of campaign objects.
 */
function getCampaignsData() {
  const rawData = getDataFromSheet('Campaigns');

  return rawData.map(row => {
    const base = {
      id: row['Campaign_ID'] || '',
      text: row['Campaign_Text'] || '',
      category: row['Target_Customer_Category'] || '',
      start: row['Campaign_Start_DateTime'] || '',
      end: row['Campaign_End_DateTime'] || '',
      count: Number(row['Campaign_Message_Count']) || 0,
      status: row['Campaign_Status'] || '',
      type: row['Campaign_Type'] || ''
    };

    // Dynamically attach up to 5 templates/timings
    for (let i = 1; i <= 5; i++) {
      base[`temp${i}`] = row[`Message_Template #${i}`] || '';
      base[`time${i}`] = row[`Message_Send_Timing #${i}`] || '';
    }

    return base;
  });
}

/**
 * Helper to normalize time values read from the sheet.
 * Converts Date objects (common for time formats like "1:50:00 PM")
 * or time strings into a standard "HH:MM" string for consistent comparison.
 * @param {any} timeValue The value read from the sheet cell.
 * @returns {string} The normalized time string (e.g., "13:50").
 */
function normalizeTime(timeValue) {
    if (timeValue instanceof Date) {
        // Extract hour and minute from the Date object provided by getValues()
        const hour = timeValue.getHours();
        const minute = timeValue.getMinutes();
        // Return in 24-hour HH:MM format (e.g., "13:50")
        return `${hour.toString().padStart(2, '0')}:${minute.toString().padStart(2, '0')}`;
    }
    // If it's already a string (e.g., "09:00:00"), sanitize it slightly for comparison
    if (typeof timeValue === 'string') {
        return timeValue.split(':').slice(0, 2).join(':');
    }
    return '';
}

/**
 * Helper function to check if the current time is ready for sending:
 * Current time (HH:MM) is greater than or equal to scheduled time (HH:MM).
 *
 * @param {string} scheduledTimeString Time in HH:MM format (e.g., "13:50").
 * @param {Date} now The current Date object (using Spreadsheet timezone).
 * @returns {boolean} True if the current time is >= the scheduled time.
 */
function isTimeReadyForSend(scheduledTimeString, now) {
    if (!scheduledTimeString || typeof scheduledTimeString !== 'string') {
        return false;
    }

    const parts = scheduledTimeString.split(':');
    if (parts.length < 2) {
        return false;
    }

    // Scheduled time components (HH:MM)
    const scheduledHour = parseInt(parts[0], 10);
    const scheduledMinute = parseInt(parts[1], 10);

    // Current time components (HH:MM)
    const currentHour = now.getHours();
    const currentMinute = now.getMinutes();

    // Convert both to a single minute value for easy comparison since they share the same date
    const scheduledMinutesToday = (scheduledHour * 60) + scheduledMinute;
    const currentMinutesToday = (currentHour * 60) + currentMinute;

    return currentMinutesToday >= scheduledMinutesToday;
}

/**
 * Implements actual SMS sending using the Twilio REST API via Google Apps Script's UrlFetchApp service.
 * Twilio Free Account Daily SMS Limit: 50
 * @param {string} msg_from The sender's phone number in E.164 format (e.g., '+919876543210').
 * @param {string} msg_to The recipient's phone number in E.164 format (e.g., '+919876543210').
 * @param {string} msg_text The message body to be sent via SMS.
 * @returns {boolean} True if the SMS was successfully triggered and accepted by Twilio, false otherwise.
 */
const SMS_MSG_FROM="+18782511984" // TWILIO sender's number
const SMS_MSG_TO="+919986758671"  // receiver's number
function trigger_sms(msg_from, msg_to, msg_text) {
  try {
    var sid = SCRIPT_PROPERTIES.getProperty('TWILIO_ACCOUNT_SID');
    var token = SCRIPT_PROPERTIES.getProperty('TWILIO_AUTH_TOKEN');
    
    var payload = {
      To: msg_to,
      From: msg_from,
      Body: msg_text
    };

    var options = {
      method: "post",
      payload: payload,
      headers: {
        Authorization: "Basic " + Utilities.base64Encode(sid + ":" + token)
      },
      muteHttpExceptions: true
    };

    var response = UrlFetchApp.fetch("https://api.twilio.com/2010-04-01/Accounts/" + sid + "/Messages.json", options);
    var result = JSON.parse(response.getContentText());

    if (result.sid) {
      Logger.log("SUCCESS: SMS sent to %s (SID: %s)", msg_to, result.sid);
      return true;
    } else if (result.code === 63038) {
      Logger.log("QUOTA ERROR: Twilio SMS daily limit hit.");
      throw new Error('QUOTA_EXCEEDED');
    } else {
      Logger.log("ERROR: Failed to send SMS to %s. Response: %s", msg_to, response.getContentText());
      return false;
    }
  } catch (e) {
    Logger.log("EXCEPTION: SMS sending failed to %s. Error: %s", msg_to, e.toString());
    return false;
  }
}

/**
 * The main function to run the campaign check.
 * - Reads campaigns and customer chats.
 * - Matches customers by chat category.
 * - Sends Email, SMS, WhatsApp messages.
 * - Manages separate per-day quotas for each channel.
 * - Does NOT update or modify any Google Sheet data.
 * This should be set up as a time-driven trigger in Google Apps Script, OR
 * can be called by doGet() when deployed as a Web App.
 *
 * Logic uses PropertiesService to track sent status for the day, preventing duplicates.
 *
 * @returns {string} A status message indicating the completion of the check.
 */
function main_campaign_runner() {
  // --- 1. Get Current Time in Spreadsheet Timezone ---
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const now = new Date();
  // Use Utilities.formatDate to adjust 'now' to the correct timezone for display/logging
  const nowInTZ = Utilities.formatDate(now, ss.getSpreadsheetTimeZone(), 'HH:mm:ss');
  Logger.log('Script execution time (Spreadsheet TZ): %s', nowInTZ);

  // --- 2. Process Campaigns ---
  const campaigns = getCampaignsData();
  const chats = getCustomerChatsData();
  let msgsTriggered = 0;
  let quotaHit = { email: false, sms: false, whatsapp: false };

  Logger.log(`Loaded ${campaigns.length} campaigns and ${chats.length} chats`);
  // campaigns.forEach(c => Logger.log(`Campaign: ${c.id}, Status=${c.status}, Category=${c.category}, Time1=${c.time1}`));
  // clients.forEach(cl => Logger.log(`Client: ${cl.name}, Category=${cl.category}, Email=${cl.email}`));

  // Use a try/catch block around the entire processing loop to catch the QUOTA_EXCEEDED flag
  try {
    campaigns.forEach(campaign => {
      // Check for required status, Only ACTIVE/UPCOMING
      if (campaign.status !== 'ACTIVE' && campaign.status !== 'UPCOMING') {
        // Logger.log(`SKIP: Campaign ${campaign.id} ignored due to status "${campaign.status}"`);
        return;
      }

      // ==============================================================================================================
      // Define the templates and timings to check based *strictly* on Campaign Message Count dynamically (1–10)
      // ==============================================================================================================
      const messages = [];

      for (let i = 1; i <= 10; i++) {
        if (campaign.count >= i) {
          let temp = campaign[`temp${i}`];
          let time = campaign[`time${i}`];

          if (temp && time) {
            messages.push({
              template: temp,
              timing: normalizeTime(time),
              number: i
            });
          }
        }
      }
      // =======================================================

      // Loop through all messages due for sending
      messages.forEach(message => {
        const { template, timing, number } = message;

        // CRITICAL STEP 1: Check if the time is ready (current time >= scheduled time)
        const normalizedTiming = normalizeTime(timing);
        if (!isTimeReadyForSend(normalizedTiming, now)) {
          Logger.log(`SKIP: Campaign ${campaign.id} Msg#${number} not yet time (${normalizedTiming})`);
          return;
        }
        
        // Already sent today?
        // It considers all mediums SMS, WhatsApp and Email
        // It considers as SENT even if partial medium were successfull in sending msg.
        if (wasSentToday(campaign.id, number)) {
          Logger.log('SKIP: Campaign %s Msg#%s already sent today.', campaign.id, number);
          return;
        }

        // Time is ready AND not sent today, trigger sms/email
        Logger.log('SENDING: Campaign %s Msg#%s (%s)', campaign.id, number, timing);

        const msg_sub = `[${campaign.id}] Campaign Message #${number} - Scheduled at ${timing}`;
        const msg_text = `Template: ${template}\nTime: ${timing}`;

        // --- Match customers by category ---
        // Campaign category must be found in one of the comma-separated words in customer's category
        const matchedCustomers = chats.filter(c => {
          if (!c.category || !campaign.category) return false;

          // Split customer's categories by comma, trim, and lowercase each one
          const customerCategories = c.category
            .toString()
            .split(',')
            .map(cat => cat.trim().toLowerCase())
            .filter(Boolean);

          const campaignCategory = campaign.category.toString().trim().toLowerCase();

          return customerCategories.includes(campaignCategory);
        });

        if (matchedCustomers.length === 0) {
          Logger.log('No customers found for category "%s"', campaign.category);
          return;
        }

        Logger.log('Matched %d customers for "%s"', matchedCustomers.length, campaign.category);

        matchedCustomers.forEach(customer => {
          if (quotaHit.email && quotaHit.sms && quotaHit.whatsapp) return; // stop all if all limits hit

          const msg_text = `Hi ${customer.name || "Customer"},\n${template}`;

          // --- Call Email Function ---
          if (!quotaHit.email && customer.email) {
            try {
              const success = trigger_email(
                EMAIL_MSG_FROM, // test sender's email address
                customer.email,
                msg_sub,
                msg_text
              );
              if (!success) Logger.log('Email failed for %s', customer.email);
            } catch (e) {
              if (e.message === 'QUOTA_EXCEEDED') {
                quotaHit.email = true;
                Logger.log('Email quota reached.');
              }
            }
          }

          // --- Call SMS Function ---
          // if (!quotaHit.sms && customer.phone) {
          //   try {
          //     const success = trigger_sms(
          //       SMS_MSG_FROM, // Twilio test sender's phone number
          //       customer.phone,
          //       msg_text
          //       );
          //     if (!success) Logger.log('SMS failed for %s', customer.phone);
          //   } catch (e) {
          //     if (e.message === 'QUOTA_EXCEEDED') {
          //       quotaHit.sms = true;
          //       Logger.log('SMS quota reached.');
          //     } else {
          //       Logger.log('SMS Error: %s', e.toString());
          //     }
          //   }
          // }

          // --- Call WhatsApp Function ---
          // if (!quotaHit.whatsapp && customer.phone) {
          //   try {
          //     const formattedPhone = 'whatsapp:' + customer.phone.replace(/\s|\+|-/g, '');
          //     const success = trigger_whatsapp_msg(
          //       WHATSAPP_MSG_FROM, // Twilio test whatsapp sender's phone number
          //       formattedPhone,
          //       msg_text
          //     );
          //     if (!success) Logger.log('WhatsApp failed for %s', customer.phone);
          //   } catch (e) {
          //     if (e.message === 'QUOTA_EXCEEDED') {
          //       quotaHit.whatsapp = true;
          //       Logger.log('WhatsApp quota reached.');
          //     } else {
          //       Logger.log('WhatsApp Error: %s', e.toString());
          //     }
          //   }
          // }
        });

        setSentLog(campaign.id, number);
        msgsTriggered++;
      });
    });
  } catch (e) {
    Logger.log('Execution stopped with error: %s', e.toString());
    throw e;
  }

  Logger.log('Completed campaigns. Messages triggered: %d', msgsTriggered);
  return `Completed at ${nowInTZ}. Messages triggered: ${msgsTriggered}`;
}

/**
 * Checks if Message Template/Timing columns exist in Campaigns sheet
 */
function ensureCampaignColumns(count) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("Campaigns");
  if (!sheet) return;

  // Normalize existing headers
  const headerRange = sheet.getRange(1, 1, 1, sheet.getLastColumn());
  let headers = headerRange.getValues()[0].map(h =>
    typeof h === "string" ? h.toString().trim().replace(/\s+/g, " ") : ""
  );

  let lastCol = sheet.getLastColumn();

  for (let i = 1; i <= count; i++) {
    const tempHeader = `Message_Template #${i}`;
    const timeHeader = `Message_Send_Timing #${i}`;

    // Check case-insensitive, trimmed
    const tempExists = headers.some(h => h.toLowerCase() === tempHeader.toLowerCase());
    const timeExists = headers.some(h => h.toLowerCase() === timeHeader.toLowerCase());

    if (!tempExists) {
      sheet.insertColumnAfter(lastCol);
      lastCol++;
      sheet.getRange(1, lastCol).setValue(tempHeader);
      headers.push(tempHeader);
    }

    if (!timeExists) {
      sheet.insertColumnAfter(lastCol);
      lastCol++;
      sheet.getRange(1, lastCol).setValue(timeHeader);
      headers.push(timeHeader);
    }
  }
}

/**
 * Web app endpoint for React frontend to add new Campaigns
 */
function doPost(e) {
  try {
    // -----------------------
    // 1️⃣ Validate POST data
    // -----------------------
    if (!e || !e.postData || !e.postData.contents) {
      return jsonResponse({ status: "error", message: "No POST data received" });
    }

    let data = {};
    try {
      if (e.postData.type === "application/json") {
        data = JSON.parse(e.postData.contents);
      } else {
        data = JSON.parse(e.postData.contents || '{}');
      }
    } catch (err) {
      return jsonResponse({ status: "error", message: "Invalid JSON: " + err.message });
    }

    // -----------------------
    // 2️⃣ Handle chat sessions
    // -----------------------
    if (data.clientId && data.chatId && data.transcriptText) {
      saveChatToSheet(data);
      return jsonResponse({ status: "ok", message: "Chat session saved" });
    }

    // -----------------------
    // 3️⃣ Handle campaigns
    // -----------------------
    saveCampaignToSheet(data);
    return jsonResponse({ status: "success", message: "Campaign saved" });

  } catch (err) {
    // -----------------------
    // 4️⃣ Global catch — always JSON
    // -----------------------
    return jsonResponse({ status: "error", message: err.message });
  }
}

// -----------------------------------------------------------
// Helper: return JSON response safely
// -----------------------------------------------------------
function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

  sheet.appendRow([
    session.clientId || "",
    session.clientName || "",
    session.clientEmail || "",
    ("'" + session.clientPhone) || "",
    session.chatId || "",
    session.date || "",
    session.time || "",
    session.transcriptText || "",
    "", "", "", ""  // empty optional fields
  ]);
}

// -----------------------------------------------------------
// Helper: Save campaign data to "Campaigns" sheet
// -----------------------------------------------------------
function saveCampaignToSheet(data) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("Campaigns");
  if (!sheet) throw new Error("Campaigns sheet not found");

  let count = Number(data.campaignMessageCount) || 0;
  if (count > 10) count = 10;

  ensureCampaignColumns(count);

  // Read headers
  let headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
  headers = headers.map(h => (typeof h === 'string' ? h.trim() : h));

  const row = new Array(headers.length).fill("");

  const fieldMap = {
    "Campaign_Text": "campaignText",
    "Target_Customer_Category": "targetCustomerCategory",
    "Campaign_Start_DateTime": "startDateTime",
    "Campaign_End_DateTime": "endDateTime",
    "Campaign_Message_Count": "campaignMessageCount",
    "Campaign_Type" : "campaignType",
    "Campaign_Status": "campaignStatus"
  };

  // Fill base fields
  headers.forEach((header, idx) => {
    if (data[header] !== undefined) {
      row[idx] = data[header];
      return;
    }
    const fk = fieldMap[header];
    if (fk && data[fk] !== undefined) {
      row[idx] = data[fk];
    }
  });

  // Fill message templates & timings
  for (let i = 1; i <= count; i++) {
    const templateKeyFrontend = `messageTemplate${i}`;
    const timingKeyFrontend = `messageSendTiming${i}`;

    const templateHeader = `Message Template #${i}`;
    const timingHeader = `Message Send Timing #${i}`;

    const templateCol = headers.indexOf(templateHeader);
    const timingCol = headers.indexOf(timingHeader);

    if (templateCol !== -1) {
      if (data[templateKeyFrontend] !== undefined) {
        row[templateCol] = data[templateKeyFrontend];
      } else if (data[templateHeader] !== undefined) {
        row[templateCol] = data[templateHeader];
      }
    }

    if (timingCol !== -1) {
      if (data[timingKeyFrontend] !== undefined) {
        row[timingCol] = data[timingKeyFrontend];
      } else if (data[timingHeader] !== undefined) {
        row[timingCol] = data[timingHeader];
      }
    }
  }

  sheet.appendRow(row);
}
